/* ===================== BREADCRUMB ===================== */
function renderBreadcrumb(){
  const el = document.getElementById('breadcrumb');
  const crumbs = [];
  const roleLabel = {brand:'มุมมองแบรนด์',owner:'มุมมองหัวหน้า',staff:'มุมมองพนักงาน'}[state.role];
  crumbs.push({label: state.role==='brand' ? 'แคมเปญของฉัน' : (state.role==='owner'?'แบรนด์ทั้งหมด':'แบรนด์ของฉัน'), action:()=>go({view:'dashboard',brandId:null,campaignId:null})});
  if(state.brandId && state.role!=='brand'){
    crumbs.push({label:getBrand(state.brandId).name, action:()=>go({view:'brand',campaignId:null})});
  }
  if(state.campaignId){
    const b = state.brandId || (state.role==='brand' ? DB.brands[0].id : null);
    const camp = getCampaign(b,state.campaignId);
    crumbs.push({label:camp.name, action:null});
  }
  el.innerHTML = crumbs.map((c,i)=>{
    const isLast = i===crumbs.length-1;
    return (i>0?'<span class="sep">›</span>':'') + `<button class="${isLast?'current':''}" ${c.action?`onclick="(${c.action.toString()})()"`:''}>${c.label}</button>`;
  }).join('');
}

    </div>
    <div class="grid cols-2">
      ${DB.brands.map(b=>{
        const found = b.campaigns.reduce((s,c)=>s+totalTier(c,'sourced'),0);

            </div>
          </div>

      }).join('')}
    </div>
  `;
}

/* ===================== EMPTY STATE ===================== */
function emptyState(icon,title,desc){
  return `<div class="card"><div class="empty"><div class="icon">${icon}</div><div class="title">${title}</div><div>${desc}</div></div></div>`;
}

/* ===================== MAIN RENDER ===================== */
function render(){
  renderRoleSwitch();
  renderBreadcrumb();
  const el = document.getElementById('content');
  if(state.role==='brand'){
    if(state.view==='dashboard') el.innerHTML = renderBrandDashboard();
    else el.innerHTML = renderBrandCampaign();
  } else if(state.role==='owner'){
    if(state.view==='dashboard') el.innerHTML = renderOwnerDashboard();
    else if(state.view==='brand') el.innerHTML = renderOwnerBrand();
    else el.innerHTML = renderOwnerCampaign();
  } else {
    if(state.view==='dashboard') el.innerHTML = renderStaffDashboard();
    else if(state.view==='brand') el.innerHTML = renderStaffBrand();
    else el.innerHTML = renderStaffCampaign();
  }
}

render();
